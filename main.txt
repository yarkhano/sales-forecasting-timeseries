import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import joblib

from statsmodels.tsa.seasonal import seasonal_decompose
from statsmodels.tsa.stattools import adfuller
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DATA_DIR = os.path.join(BASE_DIR, "data")
MODEL_DIR = os.path.join(BASE_DIR, "models")
REPORT_DIR = os.path.join(BASE_DIR, "report")
os.makedirs(DATA_DIR, exist_ok=True)
os.makedirs(MODEL_DIR, exist_ok=True)
os.makedirs(REPORT_DIR, exist_ok=True)


# ---------- 1. Mock data generation ----------
def generate_mock_data(n_days=730, seed=42):
    rng = np.random.default_rng(seed)
    dates = pd.date_range(start="2023-01-01", periods=n_days, freq="D")

    trend = np.linspace(50, 120, n_days)                          # gradual growth
    weekly_seasonality = 15 * np.sin(2 * np.pi * np.arange(n_days) / 7)  # weekly cycle
    yearly_seasonality = 10 * np.sin(2 * np.pi * np.arange(n_days) / 365) # yearly cycle
    noise = rng.normal(0, 5, n_days)

    sales = trend + weekly_seasonality + yearly_seasonality + noise
    sales = np.clip(sales, 0, None).round().astype(int)

    df = pd.DataFrame({"date": dates, "sales_units": sales})
    df.to_csv(os.path.join(DATA_DIR, "sales_data.csv"), index=False)
    print(f"Generated {len(df)} days of mock sales data.")
    return df


# ---------- 2. Visualization ----------
def plot_raw_series(df):
    plt.figure(figsize=(12, 4))
    plt.plot(df["date"], df["sales_units"], label="Daily Sales")
    plt.plot(df["date"], df["sales_units"].rolling(7).mean(), label="7-day Rolling Mean", color="red")
    plt.title("Daily Sales Over Time")
    plt.xlabel("Date")
    plt.ylabel("Sales Units")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(REPORT_DIR, "raw_series.png"))
    plt.close()


# ---------- 3. Decomposition ----------
def decompose_series(df):
    series = df.set_index("date")["sales_units"]
    decomposition = seasonal_decompose(series, model="additive", period=7)
    fig = decomposition.plot()
    fig.set_size_inches(10, 8)
    plt.tight_layout()
    plt.savefig(os.path.join(REPORT_DIR, "decomposition.png"))
    plt.close()
    return decomposition


# ---------- 4. Stationarity check ----------
def check_stationarity(series):
    result = adfuller(series)
    print(f"ADF Statistic: {result[0]:.4f}")
    print(f"p-value: {result[1]:.4f}")
    is_stationary = result[1] < 0.05
    print("Stationary" if is_stationary else "Non-stationary (differencing may help)")
    return is_stationary


# ---------- 5. Evaluation metrics ----------
def evaluate_forecast(y_true, y_pred, label=""):
    mae = mean_absolute_error(y_true, y_pred)
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    mape = np.mean(np.abs((y_true - y_pred) / np.clip(y_true, 1, None))) * 100
    print(f"\n--- {label} ---")
    print(f"MAE : {mae:.3f}")
    print(f"RMSE: {rmse:.3f}")
    print(f"MAPE: {mape:.2f}%")
    return {"mae": mae, "rmse": rmse, "mape": mape}


# ---------- 6. Models ----------
def moving_average_forecast(train, test, window=7):
    history = list(train)
    preds = []
    for _ in range(len(test)):
        preds.append(np.mean(history[-window:]))
        history.append(preds[-1])  # walk-forward using own prediction
    return np.array(preds)


def exponential_smoothing_forecast(train, test):
    model = ExponentialSmoothing(train, trend="add", seasonal="add", seasonal_periods=7)
    fit = model.fit()
    preds = fit.forecast(len(test))
    return preds.values, fit


def sarima_forecast(train, test, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7)):
    model = SARIMAX(train, order=order, seasonal_order=seasonal_order,
                     enforce_stationarity=False, enforce_invertibility=False)
    fit = model.fit(disp=False)
    forecast_result = fit.get_forecast(steps=len(test))
    preds = forecast_result.predicted_mean
    conf_int = forecast_result.conf_int()
    return preds.values, conf_int, fit


def random_forest_lag_forecast(df, n_lags=7, test_size=28):
    data = df.copy()
    for lag in range(1, n_lags + 1):
        data[f"lag_{lag}"] = data["sales_units"].shift(lag)
    data = data.dropna().reset_index(drop=True)

    feature_cols = [f"lag_{lag}" for lag in range(1, n_lags + 1)]
    X = data[feature_cols]
    y = data["sales_units"]

    X_train, X_test = X.iloc[:-test_size], X.iloc[-test_size:]
    y_train, y_test = y.iloc[:-test_size], y.iloc[-test_size:]

    model = RandomForestRegressor(n_estimators=200, random_state=42)
    model.fit(X_train, y_train)
    preds = model.predict(X_test)
    return preds, y_test.values, model


# ---------- Main ----------
if __name__ == "__main__":
    data_path = os.path.join(DATA_DIR, "sales_data.csv")

    if not os.path.exists(data_path):
        df = generate_mock_data()
    else:
        df = pd.read_csv(data_path, parse_dates=["date"])

    plot_raw_series(df)
    decomposition = decompose_series(df)

    series = df.set_index("date")["sales_units"]
    check_stationarity(series)

    # Time-based train/test split — last 28 days (4 weeks) held out
    test_size = 28
    train, test = series[:-test_size], series[-test_size:]

    results = {}

    ma_preds = moving_average_forecast(train.values, test.values)
    results["Moving Average"] = evaluate_forecast(test.values, ma_preds, "Moving Average")

    es_preds, es_model = exponential_smoothing_forecast(train, test)
    results["Exponential Smoothing"] = evaluate_forecast(test.values, es_preds, "Exponential Smoothing")

    sarima_preds, sarima_conf_int, sarima_model = sarima_forecast(train, test)
    results["SARIMA"] = evaluate_forecast(test.values, sarima_preds, "SARIMA")

    rf_preds, rf_actual, rf_model = random_forest_lag_forecast(df, n_lags=7, test_size=test_size)
    results["Random Forest (lagged)"] = evaluate_forecast(rf_actual, rf_preds, "Random Forest (lagged features)")

    # Pick best model by RMSE
    best_model_name = min(results, key=lambda k: results[k]["rmse"])
    print(f"\nBest model: {best_model_name} (RMSE={results[best_model_name]['rmse']:.3f})")

    # Plot forecast vs actual for SARIMA (has native confidence intervals)
    plt.figure(figsize=(12, 5))
    plt.plot(train.index[-60:], train.values[-60:], label="Train (last 60 days)")
    plt.plot(test.index, test.values, label="Actual", color="black")
    plt.plot(test.index, sarima_preds, label="SARIMA Forecast", color="red")
    plt.fill_between(test.index, sarima_conf_int.iloc[:, 0], sarima_conf_int.iloc[:, 1],
                      color="red", alpha=0.2, label="95% Confidence Interval")
    plt.legend()
    plt.title("SARIMA Forecast vs Actual (last 4 weeks)")
    plt.tight_layout()
    plt.savefig(os.path.join(REPORT_DIR, "forecast_vs_actual.png"))
    plt.close()

    # Forecast next 4 weeks beyond the full dataset
    full_model = SARIMAX(series, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7),
                          enforce_stationarity=False, enforce_invertibility=False).fit(disp=False)
    future_forecast = full_model.get_forecast(steps=28)
    future_preds = future_forecast.predicted_mean
    future_conf_int = future_forecast.conf_int()
    print("\nNext 4 weeks forecast:")
    print(future_preds)

    # Serialize best model (SARIMA fit object)
    joblib.dump(full_model, os.path.join(MODEL_DIR, "forecast_model.pkl"))
    print(f"\nModel saved to {os.path.join(MODEL_DIR, 'forecast_model.pkl')}")
